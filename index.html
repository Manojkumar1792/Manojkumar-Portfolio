export function buildReleaseChecklist(lockDate: Date): ChecklistItem[] {
  return TASK_OFFSETS.map(({ name, daysFromLock }) => ({
    name,
    dueDate: addDays(lockDate, daysFromLock),
    status: 'pending',
  }));
}

function addDays(date: Date, days: number): Date {
  const result = new Date(date);
  result.setDate(result.getDate() + days);     // negative offsets run before lock, positive after
  return result;
}

leaderboard.ts
export function computeLeaderboard(entries: Submission[]): LeaderboardRow[] {
  const points = new Map<string, number>();
  for (const entry of entries) {
    points.set(entry.member, (points.get(entry.member) ?? 0) + scoreFor(entry));
  }
  return [...points.entries()]
    .sort(([, a], [, b]) => b - a)               // highest points first
    .map(([member, total], i) => ({ rank: i + 1, member, total }));
}

function scoreFor(entry: Submission): number {
  const WEIGHTS: Record<Submission['type'], number> = {
    idea: 10, demo: 20, training: 5, showcase: 15,
  };
  return WEIGHTS[entry.type] ?? 0;
}

quarantine.ts
export function shouldQuarantine(history: TestRunOutcome[]): boolean {
  const recent = history.slice(-5);
  const flips = recent.filter((o, i) => i > 0 && o !== recent[i - 1]).length;
  return flips >= 2;                             // alternates pass/fail too often = flaky
}

export function classifyGate(suite: SuiteResult): 'pass' | 'block' {
  return suite.criticalFailures === 0 ? 'pass' : 'block';
}
Design Decision

"Chose a rolling 5-run history for flake detection over a single pass/fail flag — a test that alternates outcomes is a different risk than one that fails consistently, and only the former should be quarantined."

export function finalScore(scores: JudgeScore[], requiredJudges = 6): number | null {
  if (scores.length < requiredJudges) return null;   // not fully judged yet
  const total = scores.reduce((sum, s) => sum + weightedTotal(s), 0);
  return Math.round((total / scores.length) * 100) / 100;
}

function weightedTotal(score: JudgeScore): number {
  return Object.entries(score.criteria)
    .reduce((sum, [key, value]) => sum + value * CRITERIA_WEIGHTS[key], 0);
}

export function scoreSubmission(scores: JudgeDimensionScores): number {
